\newcommand\CheckRules[5]{%
  \Check{#1 single}{#2}{100pt}{}{Text k domu.}%
  \Check{#1 a}{#3}{100pt}{}{Text a domu.}%
  \Check{#1 percent}{#4}{100pt}{}{Text 50 \%.}%
  \Check{#1 section}{#5}{100pt}{}{Text § 50.}}

\newcommand\CheckToggle[5]{%
  \csname cstypo#1Enable\endcsname
  \csname cstypo#1Enable\endcsname
  \CheckRules{#1 enabled twice}{#2}{#3}{#4}{#5}%
  \csname cstypo#1Disable\endcsname
  \CheckRules{#1 disabled once}{0}{0}{0}{0}%
  \csname cstypo#1Disable\endcsname
  \CheckRules{#1 disabled twice}{0}{0}{0}{0}%
  \csname cstypo#1Enable\endcsname
  \CheckRules{#1 re-enabled}{#2}{#3}{#4}{#5}%
  \csname cstypo#1Disable\endcsname}
